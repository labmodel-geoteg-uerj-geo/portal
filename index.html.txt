<div id="results"></div>

<script src="https://unpkg.com/leaflet/dist/leaflet.js"></script>

<script>
const map = L.map('map').setView([-12, -38], 10);

L.tileLayer('https://{s}.tile.openstreetmap.org/{z}/{x}/{y}.png').addTo(map);

L.tileLayer.wms("http://207.246.112.88:8080/geoserver/ide/wms", {
    layers: 'ide:sua_tabela',
    format: 'image/png',
    transparent: true
}).addTo(map);

// catálogo
let dados = [];

fetch('data.json')
  .then(res => res.json())
  .then(data => {
    dados = data;
    mostrar(dados);
  });

function mostrar(lista) {
  const div = document.getElementById('results');
  div.innerHTML = '';

  lista.forEach(item => {
    div.innerHTML += `
      <div class="card">
        <h3>${item.titulo}</h3>
        <p><b>${item.autor}</b> (${item.ano})</p>
        <p>${item.tipo} - ${item.instituicao}</p>
        <p><i>${item.palavras_chave}</i></p>
        <p>${item.resumo}</p>
        <a href="${item.link}" target="_blank">Ver trabalho</a>
      </div>
    `;
  });
}

// busca
document.getElementById('search').addEventListener('input', e => {
  const termo = e.target.value.toLowerCase();

  const filtrado = dados.filter(d =>
    d.titulo.toLowerCase().includes(termo) ||
    d.autor.toLowerCase().includes(termo) ||
    d.palavras_chave.toLowerCase().includes(termo)
  );

  mostrar(filtrado);
});
</script>
